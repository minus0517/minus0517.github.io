#include <stdio.h>
void max_min(int[], int, int*, int*);

int main(void){
  int tab[10] = {5, 4, 9, 15, -56, -102, -1, 60, 105, 0};
  int min, max;
  max_min(tab, 10, &max, &min);
  printf("max: %d min: %d\n", max, min);
}

void max_min(int tab[], int n, int *max, int *min){
   *min = tab[0];
   *max = tab[0];
   for(int i=1; i<n; i++){
     if(tab[i] < *min) *min = tab[i];
     if(tab[i] > *max) *max = tab[i];
   }
}

int main(void){
  char napis[100];
  pierwsze_slowo(napis);
  printf("%s", napis);
  return 0;
}

void pierwsze_slowo(char *napis){
  int c;
  while((c=getchar()) != EOF && isspace(c));
  while(c != EOF && !isspace(c))
  {
    *(napis++) = (char)c;
    c = getchar();
  }
  *napis = '\0';
}

#include <stdio.h>
#define MAX_LEN 100

typedef struct Towar{
  char nazwa[MAX_LEN];
  double cenaBrutto;
  float stawkaVAT;
}Towar;

void show(const Towar *towar);
double cenaNetto(const Towar *towar);
double kwotaVAT(const Towar *towar);
Towar make_Towar(const char *nazwa, double cenaBrutto, float stawkaVAT);
Towar makeTowar(const Towar *towar);

int main(void){
  Towar a={"Wiertarka",123.00,0.23};
  Towar b={"Młot pneumatyczny", 340.00, 0.24};
  printf("%s %.2lf \n", a.nazwa, a.cenaBrutto);
  show(&a);
  show(&b);
  printf("cena Netto towaru %s wynosi %.2lf\n", a.nazwa, cenaNetto(&a));
  printf("kwota VAT towaru %s wynosi %.2lf\n", a.nazwa, kwotaVAT(&a));
  Towar c;
  c=make_Towar("Pilarka kątowa", 219.59, 0.23);
  show(&c);
  Towar d;
  d=makeTowar(&c);
  show(&d);
  return 0;
}

Towar make_Towar(const char *nazwa, double cenaBrutto, float stawkaVAT){
  Towar wynik;
  int i = 0;
  while(nazwa[i] != '\0' && i<MAX_LEN-1 ){wynik.nazwa[i]=nazwa[i], i++; }
  wynik.nazwa[i] = '\0';
  wynik.cenaBrutto = cenaBrutto;
  wynik.stawkaVAT = stawkaVAT;
  return wynik;
}

Towar makeTowar(const Towar *towar){
  Towar kopia;
  int i = 0;
  while(towar->nazwa[i] != '\0'){kopia.nazwa[i]=towar->nazwa[i], i++;}
  kopia.nazwa[i]='\0';
  kopia.cenaBrutto = towar->cenaBrutto;
  kopia.stawkaVAT = towar->stawkaVAT;
  return kopia;
}

void show(const Towar *towar){
  printf("Nazwa: %s, cena brutto: %.2lf, stawka VAT:%.2lf\n", towar->nazwa, towar->cenaBrutto, towar->stawkaVAT);
  // Dwa rodzaje wypisania towar->nazwa lub (*towar).nazwa //
  // printf("2.Nazwa: %s, cena brutto: %.2lf, stawka VAT:%.2lf\n", (*towar).nazwa, (*towar).cenaBrutto, (*towar).stawkaVAT);
}

double cenaNetto(const Towar *towar){
  return(towar->cenaBrutto) / (1.0+ towar->stawkaVAT);
}

double kwotaVAT(const Towar *towar){
  return towar->cenaBrutto - cenaNetto(towar);
}
